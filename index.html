🔐 Firestore Rules (অবশ্যই Firebase এ বসাতে হবে)

rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    match /users/{uid} {
      allow read, write: if request.auth != null && request.auth.uid == uid;
      match /joinedMatches/{mid} { allow read, write: if request.auth.uid == uid; }
    }
    match /walletRequests/{uid}/items/{id} {
      allow create: if request.auth.uid == uid;
      allow read: if request.auth.uid == uid || isAdmin();
      allow update: if isAdmin();
    }
    match /tournaments/{id} {
      allow read: if request.auth != null;
      allow write: if isAdmin();
      match /players/{pid} { allow read: if isAdmin(); allow write: if request.auth != null; }
    }
    match /admins/{uid} { allow read: if request.auth != null; allow write: if request.auth.token.email == 'salasalauddin139@gmail.com'; }
    match /banners/{id} { allow read: if true; allow write: if isAdmin(); }
    match /gameModes/{id} { allow read: if true; allow write: if isAdmin(); }
    match /results/{id} { allow read: if true; allow write: if isAdmin(); }
    match /siteSettings/main { allow read: if true; allow write: if isAdmin(); }
    match /notifications/{id} { allow read: if request.auth.uid == resource.data.uid; allow write: if isAdmin(); }
    function isAdmin() { return exists(/databases/$(database)/documents/admins/$(request.auth.uid)); }
  }
}
